cmake_minimum_required(VERSION 3.20)
project(test_support_tests LANGUAGES CXX)
enable_testing()
include("${CMAKE_CURRENT_LIST_DIR}/../test_support.cmake")
find_path(ICU_INCLUDE_DIR unicode/unistr.h REQUIRED)
add_executable(test_support_diagnostics diagnostic_tests.cpp)
target_compile_features(test_support_diagnostics PRIVATE cxx_std_23)
target_include_directories(test_support_diagnostics PRIVATE "../../foundation/src" "${ICU_INCLUDE_DIR}")
target_link_libraries(test_support_diagnostics PRIVATE enishi_test_support)
function(diagnostic_case mode)
    add_test(NAME test_support_${mode} COMMAND "${CMAKE_COMMAND}"
        "-DEXECUTABLE=$<TARGET_FILE:test_support_diagnostics>" "-DMODE=${mode}"
        "-DEXPECTED=${ARGN}" -P "${CMAKE_CURRENT_LIST_DIR}/check_diagnostics.cmake")
endfunction()
diagnostic_case(success)
diagnostic_case(result "[FAIL] diagnostic_probe" "initialize renderer" "error=7" "original cause" "outer context" "diagnostic_tests.cpp:")
diagnostic_case(empty "error=7" "Result contains no error message" "diagnostic_tests.cpp:")
diagnostic_case(boolean "boolean failure" "diagnostic_tests.cpp:")
diagnostic_case(exception "exception detail")
diagnostic_case(terminate "without an active exception")
diagnostic_case(accumulator "++evaluations == 0" "false" "diagnostic_tests.cpp:")
diagnostic_case(throw "throwing assertion" "diagnostic_tests.cpp:")
