cmake_minimum_required(VERSION 3.20)
project(effect_runtime_tests LANGUAGES CXX)
find_path(ICU_INCLUDE_DIR unicode/unistr.h REQUIRED)
add_executable(effect_runtime_tests effect_runtime_tests.cpp ../../src/effect/effect_manager.cpp)
target_include_directories(effect_runtime_tests PRIVATE ../../src ../../../engine_types/src ../../../foundation/src "${ICU_INCLUDE_DIR}")
target_compile_features(effect_runtime_tests PRIVATE cxx_std_23)
if(MSVC)
    target_compile_options(effect_runtime_tests PRIVATE /utf-8 /EHsc /W4)
else()
    target_compile_options(effect_runtime_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(ENISHI_EFFECT_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(ENISHI_EFFECT_SANITIZERS)
    if(MSVC OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Effect sanitizers require GCC or Clang with GNU-style options")
    endif()
    target_compile_options(effect_runtime_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(effect_runtime_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME effect_runtime_tests COMMAND effect_runtime_tests)
set_tests_properties(effect_runtime_tests PROPERTIES TIMEOUT 60)
