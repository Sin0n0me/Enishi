cmake_minimum_required(VERSION 3.24)
project(effect LANGUAGES CXX)
include(${CMAKE_CURRENT_LIST_DIR}/../cmake/EnishiLibrary.cmake)
include(FetchContent)
FetchContent_Declare(FOUNDATION SOURCE_DIR ${CMAKE_CURRENT_LIST_DIR}/../foundation)
FetchContent_MakeAvailable(FOUNDATION)
FetchContent_Declare(ENGINE_TYPES SOURCE_DIR ${CMAKE_CURRENT_LIST_DIR}/../engine_types)
FetchContent_MakeAvailable(ENGINE_TYPES)
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS "src/*.h" "src/*.cpp")
enishi_add_library(effect ${SOURCES})
enishi_configure_library(effect INCLUDE_DIRECTORIES src PUBLIC_DEPENDENCIES foundation engine_types)
