cmake_minimum_required(VERSION 3.20)
project(command_parser_tests LANGUAGES CXX)
add_executable(command_parser_tests command_parser_tests.cpp command_parser_property_tests.cpp
    ../../src/bypass_shell/command_parser.cpp)
target_include_directories(command_parser_tests PRIVATE ../../src)
target_compile_features(command_parser_tests PRIVATE cxx_std_23)
if(MSVC)
    target_compile_options(command_parser_tests PRIVATE /utf-8 /EHsc /W4)
else()
    target_compile_options(command_parser_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(ENISHI_PARSER_SANITIZERS "Check parser memory safety and undefined behavior" OFF)
if(ENISHI_PARSER_SANITIZERS)
    if(MSVC OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Parser sanitizers require GCC or Clang with GNU-style options")
    endif()
    target_compile_options(command_parser_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(command_parser_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME command_parser_tests COMMAND command_parser_tests)
set_tests_properties(command_parser_tests PROPERTIES TIMEOUT 60)
