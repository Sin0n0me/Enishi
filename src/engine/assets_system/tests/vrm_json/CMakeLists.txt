cmake_minimum_required(VERSION 3.20)
project(vrm_json_tests LANGUAGES CXX)
set(ASSETS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../src")
add_executable(vrm_json_tests vrm_json_tests.cpp
    "${ASSETS_ROOT}/assets_system/model/model_loader/vrm/vrm_json.cpp")
target_compile_features(vrm_json_tests PRIVATE cxx_std_23)
target_include_directories(vrm_json_tests PRIVATE "${ASSETS_ROOT}")
target_include_directories(vrm_json_tests SYSTEM PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../../../../third_party/cgltf")
if(MSVC)
    target_compile_options(vrm_json_tests PRIVATE /utf-8 /W4 /EHsc)
endif()
enable_testing()
add_test(NAME vrm_json_tests COMMAND vrm_json_tests)
set_tests_properties(vrm_json_tests PROPERTIES TIMEOUT 30)
